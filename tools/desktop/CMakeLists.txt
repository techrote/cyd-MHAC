cmake_minimum_required(VERSION 3.24)
project(MHACDesktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MHAC_SDL_SOURCE "" CACHE PATH "Exact pinned SDL2 source checkout")
if(NOT EXISTS "${MHAC_SDL_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Supply MHAC_SDL_SOURCE (see docs/visual-011.md)")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_SDL_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE SDL_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SDL_HEAD STREQUAL "5d249570393f7a37e037abf22cd6012a4cc56a71")
  message(FATAL_ERROR "Unexpected SDL2 HEAD: ${SDL_HEAD}")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
  set(SDL_LIBC ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${MHAC_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
add_executable(mhac-desktop main.cpp)
target_include_directories(mhac-desktop PRIVATE ../../include)
target_link_libraries(mhac-desktop PRIVATE SDL2::SDL2-static)
add_executable(desktop-audio-test ../../tests/desktop_audio_test.cpp)
target_include_directories(desktop-audio-test PRIVATE ../../include)
foreach(t mhac-desktop desktop-audio-test)
  if(MSVC)
    target_compile_options(${t} PRIVATE /fp:strict /UNDEBUG)
  else()
    target_compile_options(${t} PRIVATE -ffp-contract=off -UNDEBUG)
  endif()
endforeach()
enable_testing()
add_test(NAME desktop-audio COMMAND desktop-audio-test)
add_test(NAME desktop-smoke COMMAND mhac-desktop --smoke "${CMAKE_BINARY_DIR}/desktop-loop.ppm")
set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 20)
